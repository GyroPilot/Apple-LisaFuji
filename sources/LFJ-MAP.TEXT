function MapVal (c : char) : integer;
{ One run length: printable ASCII from ! upward, skipping the quote, the }
{ parentheses, the star and the braces.                                  }
var k : integer;
begin
   k := ord(c) - 33;
   if ord(c) > 42 then k := k - 4;
   if ord(c) > 123 then k := k - 1;
   if ord(c) > 125 then k := k - 1;
   MapVal := k
end;

procedure MapDraw {left, top : integer};
{ Paint the land gray with its top-left corner at (left, top), then a  }
{ frame.  The window must be the current port.                          }
var y, x, i, v : integer;
    s          : Str255;
    onLand     : boolean;
    pat        : Pattern;
    r          : Rect;
begin
   for i := 0 to 7 do
      if odd(i) then
         pat[i] := 85
      else
         pat[i] := 170;
   PenNormal;
   PenPat(pat);
   for y := 0 to mapH - 1 do begin
      MapRow(y, s);
      x := 0;
      onLand := false;
      for i := 1 to length(s) do begin
         v := MapVal(s[i]);
         if onLand and (v > 0) then begin
            MoveTo(left + x, top + y);
            LineTo(left + x + v - 1, top + y)
         end;
         x := x + v;
         onLand := not onLand
      end
   end;
   PenNormal;
   SetRect(r, left - 1, top - 1, left + mapW + 1, top + mapH + 1);
   FrameRect(r)
end;
